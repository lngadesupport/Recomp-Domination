cmake_minimum_required(VERSION 3.20)
project(GSProductionRegression LANGUAGES CXX)
set(PS2RECOMP_SOURCE "" CACHE PATH "Patched, pinned PS2Recomp source checkout")
if(NOT EXISTS "${PS2RECOMP_SOURCE}/ps2xRuntime/src/lib/gs/gs_cpu_backend.cpp")
    message(FATAL_ERROR "Set PS2RECOMP_SOURCE to the patched PS2Recomp checkout")
endif()
option(GS_TEST_SANITIZE "Instrument production renderer and regression fixtures" OFF)
find_package(Threads REQUIRED)
add_library(gs_production STATIC
    "${PS2RECOMP_SOURCE}/ps2xRuntime/src/lib/gs/gs_cpu_backend.cpp"
    "${PS2RECOMP_SOURCE}/ps2xRuntime/src/lib/gs/ps2_gs_memory.cpp")
target_include_directories(gs_production PUBLIC "${PS2RECOMP_SOURCE}/ps2xRuntime/include")
target_compile_features(gs_production PUBLIC cxx_std_20)
target_link_libraries(gs_production PUBLIC Threads::Threads)
if(MSVC)
    target_compile_options(gs_production PUBLIC /arch:AVX2)
else()
    target_compile_options(gs_production PUBLIC -mavx2)
    if(GS_TEST_SANITIZE)
        target_compile_options(gs_production PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(gs_production PUBLIC -fsanitize=address,undefined)
    endif()
endif()
enable_testing()
foreach(fixture IN ITEMS gs_clut_reload gs_upload24)
    add_executable(${fixture}_test "${CMAKE_CURRENT_LIST_DIR}/../${fixture}_test.cpp")
    target_link_libraries(${fixture}_test PRIVATE gs_production)
    add_test(NAME ${fixture} COMMAND ${fixture}_test)
    set_tests_properties(${fixture} PROPERTIES TIMEOUT 30 ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0")
endforeach()

# The actual fixture uses CBP=32. Unrelated/invalid selectors must not consume
# the bounded palette log budget or emit records for another palette.
function(add_clut_trace_case name value matches)
    add_test(NAME clut_trace_${name} COMMAND gs_clut_reload_test)
    set_tests_properties(clut_trace_${name} PROPERTIES TIMEOUT 30
        ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0;PS2_TRACE_GS_CLUT_LOADS=1;PS2_TRACE_GS_CLUT_CBP=${value}")
    if(matches)
        set_tests_properties(clut_trace_${name} PROPERTIES PASS_REGULAR_EXPRESSION "\\[gs:clut-load\\] cbp=32")
    else()
        set_tests_properties(clut_trace_${name} PROPERTIES FAIL_REGULAR_EXPRESSION "\\[gs:clut-load\\]")
    endif()
endfunction()
add_clut_trace_case(match "32" TRUE)
add_clut_trace_case(other "33" FALSE)
add_clut_trace_case(empty "" FALSE)
add_clut_trace_case(negative "-1" FALSE)
add_clut_trace_case(range "16384" FALSE)
add_clut_trace_case(overflow "999999999999999999999" FALSE)
add_clut_trace_case(suffix "32x" FALSE)
